cmake_minimum_required(VERSION 3.24)
project(StarBinariesAndroidConsumer LANGUAGES CXX)

if(NOT ANDROID)
  message(FATAL_ERROR "This consumer requires an Android NDK toolchain")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Release|Debug)$")
  message(FATAL_ERROR "Pass CMAKE_BUILD_TYPE=Release or Debug")
endif()
if(NOT IS_ABSOLUTE "${STAR_SDK_ROOT}")
  message(FATAL_ERROR "Pass -DSTAR_SDK_ROOT=<absolute extracted SDK path>")
endif()
add_executable(android_consumer main.cpp ../consumer/main.cpp)
set_source_files_properties(../consumer/main.cpp PROPERTIES
  COMPILE_DEFINITIONS "main=star_smoke_main")
target_compile_features(android_consumer PRIVATE cxx_std_20)
include(../consumer/boost.cmake)
star_test_boost(android_consumer "${STAR_SDK_ROOT}" STATIC_LIBRARY "${CMAKE_BUILD_TYPE}")
include(../consumer/stdexec.cmake)
star_test_stdexec(android_consumer "${STAR_SDK_ROOT}")
include(../consumer/gsl.cmake)
star_test_gsl(android_consumer "${STAR_SDK_ROOT}")
