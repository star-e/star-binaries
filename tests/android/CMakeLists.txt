cmake_minimum_required(VERSION 3.24)
project(StarBinariesAndroidConsumer LANGUAGES CXX)

if(NOT ANDROID)
  message(FATAL_ERROR "This consumer requires an Android NDK toolchain")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Release|Debug)$")
  message(FATAL_ERROR "Pass CMAKE_BUILD_TYPE=Release or Debug")
endif()
find_package(ZLIB CONFIG REQUIRED COMPONENTS static PATHS "${CMAKE_PREFIX_PATH}/share/zlib"
  NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
string(TOUPPER "${CMAKE_BUILD_TYPE}" configuration_upper)
get_target_property(selected_library ZLIB::ZLIBSTATIC "IMPORTED_LOCATION_${configuration_upper}")
get_target_property(include_paths ZLIB::ZLIBSTATIC INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(library_type ZLIB::ZLIBSTATIC TYPE)
if(NOT library_type STREQUAL "STATIC_LIBRARY" OR NOT EXISTS "${selected_library}")
  message(FATAL_ERROR "Missing static zlib for ${CMAKE_BUILD_TYPE}")
endif()
foreach(dependency_path IN ITEMS ${include_paths} "${selected_library}")
  cmake_path(IS_PREFIX CMAKE_PREFIX_PATH "${dependency_path}" NORMALIZE from_sdk)
  if(NOT from_sdk)
    message(FATAL_ERROR "Dependency resolved outside the SDK: ${dependency_path}")
  endif()
endforeach()

add_executable(android_consumer main.cpp ../consumer/main.cpp)
set_source_files_properties(../consumer/main.cpp PROPERTIES
  COMPILE_DEFINITIONS "main=star_zlib_roundtrip")
target_compile_features(android_consumer PRIVATE cxx_std_20)
target_link_libraries(android_consumer PRIVATE ZLIB::ZLIBSTATIC)
